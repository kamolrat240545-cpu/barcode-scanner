/* ===================================================== SKU BARCODE FORMATS รองรับ Barcode แบบเส้น / 1D ===================================================== */ const SKU_FORMATS = [ // CODE 128 Html5QrcodeSupportedFormats.CODE_128, // CODE 39 Html5QrcodeSupportedFormats.CODE_39, // CODE 93 Html5QrcodeSupportedFormats.CODE_93, // EAN Html5QrcodeSupportedFormats.EAN_13, Html5QrcodeSupportedFormats.EAN_8, // UPC Html5QrcodeSupportedFormats.UPC_A, Html5QrcodeSupportedFormats.UPC_E, // ITF Html5QrcodeSupportedFormats.ITF, // CODABAR Html5QrcodeSupportedFormats.CODABAR ]; /* ===================================================== SKU CAMERA รองรับ Barcode แบบเส้นยาว เช่น CODE 128 ===================================================== */ async function startSkuScanner(){ if(skuScanning)return; skuProcessing = false; document .getElementById("sku-camera-area") .classList.remove("hidden"); document .getElementById("openSkuBtn") .classList.add("hidden"); try{ skuScanner = new Html5Qrcode("sku-reader"); skuScanning = true; await skuScanner.start( { facingMode:"environment" }, { /* * จำนวนเฟรมต่อวินาที */ fps:10, /* * กรอบสแกนสำหรับ Barcode แบบเส้น * * Barcode ของคุณเป็นแนวนอน * จึงใช้กรอบ "กว้าง + เตี้ย" */ qrbox:function(w,h){ const width = Math.floor(w * 0.92); const height = Math.max( 120, Math.min( Math.floor(h * 0.32), 240 ) ); return { width:width, height:height }; }, /* * กล้องเป็นแนวนอน 16:9 */ aspectRatio:16/9, /* * ใช้เฉพาะ Barcode แบบเส้น */ formatsToSupport:SKU_FORMATS, /* * ไม่ใช้ BarcodeDetector * ให้ html5-qrcode ใช้ ZXing * ซึ่งเหมาะกับ Barcode 1D */ useBarCodeDetectorIfSupported:false }, /* * สแกนสำเร็จ */ function(decodedText, decodedResult){ handleSkuScan(decodedText); }, /* * ระหว่างยังอ่าน Barcode ไม่ได้ * ไม่ต้องแสดง Error */ function(errorMessage){ // ไม่ต้องทำอะไร } ); /* * พยายามเปิด Continuous Autofocus * สำหรับมือถือที่รองรับ */ try{ const capabilities = skuScanner.getRunningTrackCapabilities(); if( capabilities && capabilities.focusMode && capabilities.focusMode.includes("continuous") ){ await skuScanner.applyVideoConstraints({ focusMode:"continuous" }); } }catch(e){ /* * มือถือบางรุ่นไม่รองรับ * ไม่ต้องทำอะไร */ } }catch(error){ skuScanning = false; document .getElementById("sku-camera-area") .classList.add("hidden"); document .getElementById("openSkuBtn") .classList.remove("hidden"); showMessage( "sku-message", "error", "❌ " + escapeHtml( error.message || error ) ); } } /* ===================================================== SKU SCAN ===================================================== */ async function handleSkuScan(decodedText){ if(skuProcessing)return; skuProcessing = true; const sku = String(decodedText || "").trim(); if(!sku){ skuProcessing = false; return; } /* * สแกนเจอแล้วหยุดกล้องทันที */ await stopSkuScanner(); /* * ใส่ค่าที่สแกนได้ลงช่อง SKU */ document .getElementById("skuInput") .value = sku; /* * เพิ่ม SKU เข้า List ทันที */ addSku(sku); } /* ===================================================== STOP SKU CAMERA ===================================================== */ async function stopSkuScanner(){ skuScanning = false; if(skuScanner){ try{ await skuScanner.stop(); }catch(e){} try{ await skuScanner.clear(); }catch(e){} skuScanner = null; } document .getElementById("sku-camera-area") .classList.add("hidden"); document .getElementById("openSkuBtn") .classList.remove("hidden"); }
